find_package(wxWidgets 3.2 REQUIRED COMPONENTS core base net)
include(${wxWidgets_USE_FILE})

add_executable(kicad-bscan WIN32
    app.cpp
    chain_table.cpp
    extest_dialogs.cpp
    interconnect_page.cpp
    bsdl_dialogs.cpp
    chain_page.cpp
    chip_canvas.cpp
    chip_view.cpp
    main_frame.cpp
    pins_page.cpp
    session.cpp
    extest_banner.cpp
    extest_controller.cpp
    gui_events.cpp
    single_instance.cpp)

if(WIN32)
    # Application manifest (common controls v6, DPI awareness); see the .rc file
    target_sources(kicad-bscan PRIVATE kicad-bscan.rc)
    # The .rc file refers to icons/kicad-bscan.ico (Explorer icon of the .exe)
    target_include_directories(kicad-bscan PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
    set_property(SOURCE kicad-bscan.rc APPEND PROPERTY OBJECT_DEPENDS
                 "${CMAKE_CURRENT_SOURCE_DIR}/icons/kicad-bscan.ico")
endif()

target_link_libraries(kicad-bscan PRIVATE bscan_kicad bscan_core ${wxWidgets_LIBRARIES})

# Window icon: icons/kicad-bscan-<size>.png (exported from art/kicad-bscan.svg) embedded
# as byte arrays, so no file has to be found at run time
set(BSCAN_ICON_SIZES 16 24 32 48 64 128 256)
set(_icons_h "${CMAKE_CURRENT_BINARY_DIR}/app_icons.h")
set(_content "// Generated by src/gui/CMakeLists.txt from src/gui/icons/*.png\n#pragma once\n#include <cstddef>\n\n")
set(_list "")
foreach(n ${BSCAN_ICON_SIZES})
    set(_png "${CMAKE_CURRENT_SOURCE_DIR}/icons/kicad-bscan-${n}.png")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${_png}")
    file(READ "${_png}" _hex HEX)
    string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," _hex "${_hex}")
    string(APPEND _content "static const unsigned char s_icon${n}[] = { ${_hex} };\n")
    string(APPEND _list "    { s_icon${n}, sizeof( s_icon${n} ) },\n")
endforeach()
string(APPEND _content "\nstruct AppIcon\n{\n    const unsigned char* data;\n    size_t               size;\n};\n\n"
                       "static const AppIcon s_appIcons[] = {\n${_list}};\n")
file(WRITE "${_icons_h}.tmp" "${_content}")
configure_file("${_icons_h}.tmp" "${_icons_h}" COPYONLY)   # rewritten only if changed

# Button symbols: icons/<name>.svg embedded the same way, as s_svg_<name>
set(BSCAN_SVG_ICONS pin pin-off-light pin-off-dark)
set(_svg_h "${CMAKE_CURRENT_BINARY_DIR}/svg_icons.h")
set(_content "// Generated by src/gui/CMakeLists.txt from src/gui/icons/*.svg\n#pragma once\n\n")
foreach(name ${BSCAN_SVG_ICONS})
    set(_svg "${CMAKE_CURRENT_SOURCE_DIR}/icons/${name}.svg")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${_svg}")
    file(READ "${_svg}" _hex HEX)
    string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," _hex "${_hex}")
    string(MAKE_C_IDENTIFIER "s_svg_${name}" _id)
    string(APPEND _content "static const unsigned char ${_id}[] = { ${_hex} };\n")
endforeach()
file(WRITE "${_svg_h}.tmp" "${_content}")
configure_file("${_svg_h}.tmp" "${_svg_h}" COPYONLY)
target_include_directories(kicad-bscan PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_definitions(kicad-bscan PRIVATE BSCAN_VERSION="${PROJECT_VERSION}")

if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(kicad-bscan PRIVATE -Wall -Wextra)
endif()

# ---------------------------------------------------------------------------
# Installation as a KiCad IPC plugin:
#   <KICAD_PLUGIN_DIR>/<BSCAN_PLUGIN_ID>/plugin.json
#                                       /kicad-bscan[.exe]
#                                       /icons/*.png
# ---------------------------------------------------------------------------
set(BSCAN_ENTRYPOINT "kicad-bscan${CMAKE_EXECUTABLE_SUFFIX}")
configure_file(${PROJECT_SOURCE_DIR}/plugin/plugin.json.in
               ${PROJECT_BINARY_DIR}/plugin.json @ONLY)

set(_dest "${KICAD_PLUGIN_DIR}/${BSCAN_PLUGIN_ID}")
if(WIN32)
    # Windows has no library search path like Linux: the DLLs the program needs
    # (wxWidgets, protobuf/Abseil, NNG, compiler runtime, ...) are copied next to
    # the .exe. They are looked up in PATH, in <prefix>/bin of CMAKE_PREFIX_PATH
    # (vcpkg, own installs) and in the wxWidgets library folder; DLLs of
    # Windows itself are left out.
    option(BSCAN_INSTALL_DLLS "Install the DLLs kicad-bscan.exe needs next to it" ON)
endif()

if(WIN32 AND BSCAN_INSTALL_DLLS)
    if(CMAKE_VERSION VERSION_LESS 3.21)
        message(FATAL_ERROR "Installing the DLLs needs CMake 3.21 or newer "
                            "(or -DBSCAN_INSTALL_DLLS=OFF and copy them yourself)")
    endif()

    set(_dll_dirs "")
    foreach(p ${CMAKE_PREFIX_PATH})
        list(APPEND _dll_dirs "${p}/bin")
    endforeach()
    if(wxWidgets_LIB_DIR)
        list(APPEND _dll_dirs "${wxWidgets_LIB_DIR}")
    endif()
    get_filename_component(_compiler_dir "${CMAKE_CXX_COMPILER}" DIRECTORY)
    list(APPEND _dll_dirs "${_compiler_dir}")   # MinGW runtime (libstdc++, ...)
    file(TO_CMAKE_PATH "$ENV{PATH}" _path_dirs)
    list(APPEND _dll_dirs ${_path_dirs})

    install(TARGETS kicad-bscan
            RUNTIME_DEPENDENCIES
                PRE_EXCLUDE_REGEXES "^api-ms-" "^ext-ms-"
                POST_EXCLUDE_REGEXES "[/\\\\][Ww][Ii][Nn][Dd][Oo][Ww][Ss][/\\\\]"
                DIRECTORIES ${_dll_dirs}
            RUNTIME DESTINATION "${_dest}")
else()
    install(TARGETS kicad-bscan RUNTIME DESTINATION "${_dest}")
endif()
install(FILES ${PROJECT_BINARY_DIR}/plugin.json DESTINATION "${_dest}")
install(DIRECTORY ${PROJECT_SOURCE_DIR}/plugin/icons DESTINATION "${_dest}")

# Linux desktops on Wayland (GNOME in particular) take the taskbar icon from a
# .desktop file named after the program (app id "kicad-bscan"), not from the
# window. Hidden from menus (NoDisplay): the program is started by KiCad.
if(UNIX AND NOT APPLE)
    option(BSCAN_INSTALL_DESKTOP_FILE "Install a hidden .desktop file and icons for the taskbar" ON)

    if(BSCAN_INSTALL_DESKTOP_FILE)
        if(DEFINED ENV{XDG_DATA_HOME} AND NOT "$ENV{XDG_DATA_HOME}" STREQUAL "")
            set(_xdg_data "$ENV{XDG_DATA_HOME}")
        else()
            set(_xdg_data "$ENV{HOME}/.local/share")
        endif()

        set(BSCAN_EXEC "${_dest}/kicad-bscan")
        configure_file(${PROJECT_SOURCE_DIR}/plugin/kicad-bscan.desktop.in
                       ${PROJECT_BINARY_DIR}/kicad-bscan.desktop @ONLY)
        install(FILES ${PROJECT_BINARY_DIR}/kicad-bscan.desktop
                DESTINATION "${_xdg_data}/applications")

        foreach(n ${BSCAN_ICON_SIZES})
            install(FILES ${CMAKE_CURRENT_SOURCE_DIR}/icons/kicad-bscan-${n}.png
                    DESTINATION "${_xdg_data}/icons/hicolor/${n}x${n}/apps"
                    RENAME kicad-bscan.png)
        endforeach()
        install(FILES ${PROJECT_SOURCE_DIR}/art/kicad-bscan.svg
                DESTINATION "${_xdg_data}/icons/hicolor/scalable/apps")
    endif()
endif()
